```python
def system_entropy(probability_distribution):
    """
    Shannon entropy: S = -Σ p_i log(p_i)
    
    High entropy = maximum uncertainty
    Low entropy = predictable
    """
    p = probability_distribution
    p = p[p > 0]  # Avoid log(0)
    S = -np.sum(p * np.log(p))
    return S

def entropy_production_rate(state_trajectory):
```

```python
class HysteresisLoop:
    """
    Universal hysteresis model
    
    State depends on:
    1. Current input
    2. History (which branch you're on)
    3. Path taken to get here
    """
    
    def __init__(self):
        self.upper_branch = self._upper_curve
        self.lower_branch = self._lower_curve
        self.current_branch = "lower"
        self.history = []
    
    def response(self, input_value):
        """
        Same input → different output depending on history
        """
        if self.current_branch == "lower":
            output = self.lower_branch(input_value)
            
            # Check if should jump to upper branch
            if input_value > UPPER_THRESHOLD:
                self.current_branch = "upper"
                output = self.upper_branch(input_value)
                self.history.append(("jump_up", input_value))
        
        else:  # upper branch
            output = self.upper_branch(input_value)
            
            # Check if should jump to lower branch
            if input_value < LOWER_THRESHOLD:
                self.current_branch = "lower"
                output = self.lower_branch(input_value)
                self.history.append(("jump_down", input_value))
        
        return output
    
    def hysteresis_width(self):
        """
        How much further down must you go to recover?
        
        UPPER_THRESHOLD (where you break) ≠ LOWER_THRESHOLD (where you recover)
        
        Width = UPPER_THRESHOLD - LOWER_THRESHOLD
        
        Large width = hard to recover
        """
        return UPPER_THRESHOLD - LOWER_THRESHOLD
```

```python
def cascade_dynamics(network, initial_failure, threshold_function):
    """
    Universal cascade model
    
    1. Node fails
    2. Neighbors feel increased load
    3. Some neighbors cross threshold → fail
    4. Their neighbors feel increased load
    5. CASCADE or CONTAINMENT
    
    SAME for:
    - Viral spread
    - Bank runs  
    - Panic cascades
    - Revolutions
    - Ecosystem collapse
    """
    failed = {initial_failure}
    active = {initial_failure}
    
    while active:
        newly_failed = set()
        
        for node in active:
            neighbors = network.neighbors(node)
            
            for neighbor in neighbors:
                if neighbor in failed:
                    continue
                
                # Compute increased load from failed neighbors
                failed_neighbor_fraction = sum(
                    1 for n in network.neighbors(neighbor)
                    if n in failed
                ) / network.degree(neighbor)
                
                # Check threshold
                if failed_neighbor_fraction > threshold_function(neighbor):
                    newly_failed.add(neighbor)
        
        if not newly_failed:
            break  # Cascade stopped
        
        failed.update(newly_failed)
        active = newly_failed
    
    return len(failed)  # Cascade size

# Power law emerges when system is at critical point
# P(cascade_size) ~ size^(-α)
# 
# Means: Many small cascades, few huge ones
# NO characteristic scale (scale-free)
```

```python
def percolation_threshold(network):
```
