    def classify_observation(
        self,
        observation_data: Dict[str, any]
    ) -> PhenomenologicalMetadata:
        """
        Automatically classify an observation using Dewey system
        """
        metadata = PhenomenologicalMetadata(
            observation_id=self._generate_observation_id(),
            timestamp=datetime.now()
        )
        
        # Determine substrate type
        substrate = observation_data.get('substrate_type', 'unknown')
        metadata.substrate_type = substrate
        
        # Determine primary Dewey classification based on substrate
        primary_dewey_map = {
            'individual': DeweyClass.STRESS_PSYCHOLOGY.value,
            'relationship': DeweyClass.SOCIAL_INTERACTION.value,
            'team': DeweyClass.ASSOCIATIONS.value,
            'organization': DeweyClass.MANAGEMENT.value,
            'institution': DeweyClass.PUBLIC_ADMINISTRATION.value,
            'nation': DeweyClass.POLITICAL_SCIENCE.value,
            'platform': DeweyClass.NETWORK_TOPOLOGY.value,
            'market': DeweyClass.ECONOMICS.value,
        }
        
        metadata.primary_dewey = primary_dewey_map.get(substrate, "003.7")
        
        # Add secondary classifications based on observed primitives
        for primitive, value in observation_data.get('primitives', {}).items():
            if primitive in self.primitive_to_dewey:
                metadata.secondary_dewey.extend(
                    self.primitive_to_dewey[primitive]
                )
        
        # Remove duplicates, keep primary separate
        metadata.secondary_dewey = list(set(metadata.secondary_dewey))
        if metadata.primary_dewey in metadata.secondary_dewey:
            metadata.secondary_dewey.remove(metadata.primary_dewey)
        
        # Populate primitive observations
        metadata.primitives_observed = observation_data.get('primitives', {})
        
        # Add contextual information
        metadata.system_id = observation_data.get('system_id', '')
        metadata.system_name = observation_data.get('system_name', '')
        metadata.scale = observation_data.get('scale', 'meso')
        metadata.temporal_scale = observation_data.get('temporal_scale', 'days')
        
        return metadata
    
    def _generate_observation_id(self) -> str:
        """Generate unique observation ID"""
        from uuid import uuid4
        return f"OBS-{datetime.now().strftime('%Y%m%d')}-{uuid4().hex[:8]}"
    
    def find_cross_domain_patterns(
        self,
        dewey_code: str,
        observations: List[PhenomenologicalMetadata]
    ) -> List[PhenomenologicalMetadata]:
        """
        Find patterns across different domains with same Dewey classification
        
        E.g., "catastrophic fold" appears in:
        - 616.89 (mental breakdown)
        - 330.9 (economic crash)
        - 320.9 (revolution)
        
        Returns observations from all domains sharing this pattern
        """
        matching = []
        
        for obs in observations:
            if (obs.primary_dewey.startswith(dewey_code) or
                any(d.startswith(dewey_code) for d in obs.secondary_dewey)):
                matching.append(obs)
        
        return matching

class DeweyIndexingService:
    """
    Maintains Dewey Decimal index across all observations
    Enables cross-domain pattern discovery
    """
    
    def __init__(self):
        self.firestore_client = firestore.Client()
        self.bigquery_client = bigquery.Client()
    
    def index_by_dewey(self, observation_id: str, dewey_codes: List[str]):
        """
        Create bidirectional index:
        - Observation → Dewey codes
        - Dewey code → Observations
        """
        
        # Store in dedicated Dewey index collection
        for dewey_code in dewey_codes:
            # Get or create Dewey code document
            dewey_ref = self.firestore_client.collection('dewey_index').document(
                dewey_code.replace('.', '_')  # Firestore-safe key
            )
            
            # Use array union to add observation (avoids duplicates)
            dewey_ref.set({
                'dewey_code': dewey_code,
                'observation_ids': firestore.ArrayUnion([observation_id]),
                'last_updated': firestore.SERVER_TIMESTAMP,
                'observation_count': firestore.Increment(1)
            }, merge=True)
    
    def find_cross_domain_analogues(
        self,
        observation_id: str,
        max_results: int = 10
    ) -> List[Dict]:
        """
        Find analogous patterns in different domains
        
        E.g., if observing individual breakdown (Dewey 157),

        - Team dissolution (Dewey 302.3)
        - Company bankruptcy (Dewey 330.9)
        - Nation collapse (Dewey 320.9)
        
        All sharing "catastrophic fold" primitive
        """
        
        # Get observation's Dewey codes

        obs = obs_ref.get().to_dict()
        
        primary_dewey = obs['primary_dewey']
        secondary_dewey = obs['secondary_dewey']
        
        analogues = []
        
        # For each Dewey code, find other observations
        for dewey_code in secondary_dewey:
            # Query Dewey index
            dewey_ref = self.firestore_client.collection('dewey_index').document(
                dewey_code.replace('.', '_')
            )
            
            dewey_doc = dewey_ref.get()
            
            if dewey_doc.exists:
                related_ids = dewey_doc.to_dict().get('observation_ids', [])
                
                # Retrieve those observations (exclude self)
                for related_id in related_ids[:max_results]:
                    if related_id != observation_id:
                        related_ref = self.firestore_client.collection('observations').document(related_id)
                        related = related_ref.get()
                        
                        if related.exists:
                            related_data = related.to_dict()
                            
                            # Only include if different substrate (cross-domain)
                            if related_data['substrate_type'] != obs['substrate_type']:
                                analogues.append({
                                    'observation_id': related_id,
                                    'substrate_type': related_data['substrate_type'],
                                    'shared_dewey': dewey_code,
                                    'shared_primitive': self._dewey_to_primitive(dewey_code),
                                    'similarity_score': self._compute_similarity(obs, related_data)
                                })
        
        # Sort by similarity
        analogues.sort(key=lambda x: x['similarity_score'], reverse=True)
        
        return analogues[:max_results]
    
    def _dewey_to_primitive(self, dewey_code: str) -> str:
        """Map Dewey code back to primitive name"""
        # Reverse lookup in DeweyMapper
        mapper = DeweyMapper()
        for primitive, codes in mapper.primitive_to_dewey.items():
            if dewey_code in codes:
                return primitive
        return "unknown"
    
    def _compute_similarity(self, obs1: Dict, obs2: Dict) -> float:
        """Compute similarity score between observations"""
        # Compare primitives, compute cosine similarity or other metric
        primitives1 = obs1.get('primitives_observed', {})
        primitives2 = obs2.get('primitives_observed', {})
        
        # Simple Jaccard similarity on keys
        keys1 = set(primitives1.keys())
        keys2 = set(primitives2.keys())
        
        if not keys1 or not keys2:
            return 0.0
        
        intersection = len(keys1 & keys2)
        union = len(keys1 | keys2)
        
        return intersection / union if union > 0 else 0.0
